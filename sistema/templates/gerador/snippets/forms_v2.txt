class BootstrapModelForm(forms.ModelForm):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        for field in self.fields.values():
            if isinstance(field.widget, forms.CheckboxInput): field.widget.attrs.setdefault("class", "form-check-input")
            elif isinstance(field.widget, (forms.Select, forms.SelectMultiple)): field.widget.attrs.setdefault("class", "form-select")
            else: field.widget.attrs.setdefault("class", "form-control")
            if isinstance(field, forms.ModelChoiceField): field.empty_label = "--- Selecione uma opção ---"
