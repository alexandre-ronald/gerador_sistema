def _allowed_items(request, items):
    user = getattr(request, "user", None)
    if user is None or not user.is_authenticated:
        return []
    return [item for item in items if not item["permission"] or user.has_perm(item["permission"])]


def _mark_active_items(request, items):
    resolver_match = getattr(request, "resolver_match", None)
    current_url_name = getattr(resolver_match, "url_name", None)
    current_app_name = getattr(resolver_match, "app_name", None)
    marked = []
    for item in items:
        active_names = item.get("active_url_names") or [item["url_name"].split(":")[-1]]
        marked.append({
            **item,
            "is_active": current_app_name == item.get("app_name") and current_url_name in active_names,
        })
    return marked
