def _rbac_item_allowed(request, item):
    """Aplica no menu a mesma decisão usada pelas rotas geradas."""
    entity_name = item.get("entity_name")
    if item.get("is_advanced_page") and item.get("advanced_context_kind") == "none":
        return True
    if not entity_name:
        return True
    app_name = item.get("app_name")
    if not app_name:
        return False
    try:
        rbac = importlib.import_module(f"{app_name}.rbac")
        if item.get("is_report"):
            return bool(rbac.can_report(request.user, entity_name, item.get("report_id")))
        return bool(rbac.has_entity_action(request.user, entity_name, item.get("rbac_action") or "list"))
    except Exception:
        return False


def _allowed_items(request, items):
    user = getattr(request, "user", None)
    if user is None or not user.is_authenticated:
        return []
    result = []
    for item in items:
        if item["permission"] and not user.has_perm(item["permission"]):
            continue
        if not _rbac_item_allowed(request, item):
            continue
        result.append(item)
    return result


def _mark_active_items(request, items):
    resolver_match = getattr(request, "resolver_match", None)
    current_url_name = getattr(resolver_match, "url_name", None)
    current_app_name = getattr(resolver_match, "app_name", None)
    marked = []
    for item in items:
        active_names = item.get("active_url_names") or [item["url_name"].split(":")[-1]]
        if item.get("is_advanced_page"):
            is_active = current_url_name in active_names
        else:
            is_active = current_app_name == item.get("app_name") and current_url_name in active_names
        marked.append({**item, "is_active": is_active})
    return marked

def _relation_filter_choices(request):
    resolver_match = getattr(request, "resolver_match", None)
    if not str(getattr(resolver_match, "url_name", "") or "").endswith("_list"): return []
    user = getattr(request, "user", None)
    if user is None or not user.is_authenticated: return []
    item = _current_navigation_item(request)
    if not item or item.get("is_notifications") or item.get("is_report") or item.get("is_advanced_page"): return []
    try: model = apps.get_model(item["app_name"], item["model_name"])
    except LookupError: return []
    if model is None: return []
    result = []
    for field in model._meta.fields:
        if not getattr(field, "is_relation", False) or (not getattr(field, "many_to_one", False) and not getattr(field, "one_to_one", False)): continue
        related_model = getattr(field.remote_field, "model", None)
        if related_model is None: continue
        options = [{"value": str(obj.pk), "label": str(obj)} for obj in related_model._default_manager.all()[:500]]
        options.sort(key=lambda option: option["label"].casefold())
        result.append({"field": field.name, "options": options})
    return result

def _workspace_projection(request):
    config = WORKSPACES if isinstance(WORKSPACES, dict) else {}
    raw_workspaces = config.get("workspaces") if isinstance(config.get("workspaces"), list) else []
    if not raw_workspaces:
        return {"configured": False, "workspaces": [], "default_workspace": ""}
    item_index = _navigation_item_index(request)
    visible = []
    for workspace in raw_workspaces:
        if workspace.get("enabled") is False:
            continue
        sections = []
        visible_item_ids = set()
        for section in workspace.get("sections") or []:
            if section.get("enabled") is False:
                continue
            items = []
            for configured_item in section.get("items") or []:
                if configured_item.get("enabled") is False:
                    continue
                destination = configured_item.get("destination") or {}
                key = (destination.get("kind"), destination.get("ref"))
                nav_item = item_index.get(key)
                if not nav_item:
                    continue
                projected = {
                    **nav_item,
                    "workspace_id": workspace.get("id"),
                    "workspace_label": workspace.get("label"),
                    "workspace_section_id": section.get("id"),
                    "workspace_section_label": section.get("label"),
                    "workspace_item_id": configured_item.get("id"),
                    "workspace_destination_kind": destination.get("kind"),
                    "label": configured_item.get("label") or nav_item.get("label"),
                    "is_report": False if destination.get("kind") == "report" else nav_item.get("is_report", False),
                    "group_label": "" if destination.get("kind") == "report" else nav_item.get("group_label", ""),
                }
                items.append(projected)
                visible_item_ids.add(configured_item.get("id"))
            if items:
                sections.append({"id": section.get("id"), "label": section.get("label"), "items": items})
        if not sections:
            continue
        home = workspace.get("home") if workspace.get("home") in visible_item_ids else sections[0]["items"][0].get("workspace_item_id")
        visible.append({"id": workspace.get("id"), "label": workspace.get("label"), "description": workspace.get("description"), "icon": workspace.get("icon"), "home": home, "sections": sections})
    default_workspace = config.get("default_workspace")
    visible_ids = {workspace.get("id") for workspace in visible}
    if default_workspace not in visible_ids:
        default_workspace = visible[0].get("id") if visible else ""
    return {"configured": True, "workspaces": visible, "default_workspace": default_workspace}


def _workspace_navigation_modules(workspace_projection):
    """Adapta todos os Workspaces visíveis ao contrato de menu consumido pelo base.html."""
    if not workspace_projection.get("configured"):
        return None
    modules = []
    for workspace in workspace_projection.get("workspaces") or []:
        for section in workspace.get("sections") or []:
            items = section.get("items") or []
            if items:
                modules.append({
                    "app_name": f"workspace:{workspace.get('id')}:{section.get('id')}",
                    "label": f"{workspace.get('label') or 'Workspace'} · {section.get('label') or 'Seção'}",
                    "items": items,
                    "workspace_id": workspace.get("id"),
                    "workspace_label": workspace.get("label"),
                    "workspace_section_id": section.get("id"),
                    "workspace_section_label": section.get("label"),
                })
    return modules

def _workspace_navigation_context(request, workspace_projection):
    """Resolve o contexto do item ativo entre todos os Workspaces visíveis."""
    requested_workspace = str(request.GET.get("workspace") or "").strip()
    requested_item = str(request.GET.get("workspace_item") or "").strip()
    workspaces = workspace_projection.get("workspaces") or []

    if requested_workspace or requested_item:
        if not requested_workspace or not requested_item:
            return None
        for workspace in workspaces:
            if workspace.get("id") != requested_workspace:
                continue
            for section in workspace.get("sections") or []:
                for item in section.get("items") or []:
                    if item.get("workspace_item_id") != requested_item:
                        continue
                    if not item.get("is_active"):
                        return None
                    return _workspace_context_payload(workspace, section, item)
        return None

    for workspace in workspaces:
        for section in workspace.get("sections") or []:
            for item in section.get("items") or []:
                if item.get("is_active"):
                    return _workspace_context_payload(workspace, section, item)
    return None


def _workspace_current_navigation(workspace_context, fallback):
    """Aplica o contexto do Workspace sem alterar a URL canônica da experiência."""
    if not workspace_context:
        return fallback
    section_label = workspace_context.get("section_label") or ""
    item_label = workspace_context.get("item_label") or ""
    return {
        "module_label": workspace_context.get("workspace_label") or fallback.get("module_label"),
        "item_label": f"{section_label} › {item_label}" if section_label else item_label,
        "workspace_label": workspace_context.get("workspace_label"),
        "section_label": section_label,
        "workspace_item_label": item_label,
    }
