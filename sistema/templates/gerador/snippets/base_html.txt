    <script>
        const themeToggle = document.getElementById('theme-toggle');
        const themeIcon = document.getElementById('theme-icon');
        const htmlElement = document.documentElement;
        function applyTheme(theme) {
            htmlElement.setAttribute('data-bs-theme', theme);
            localStorage.setItem('theme', theme);
            if (theme === 'dark') themeIcon.classList.replace('bi-moon-stars-fill', 'bi-sun-fill');
            else themeIcon.classList.replace('bi-sun-fill', 'bi-moon-stars-fill');
        }
        const savedTheme = localStorage.getItem('theme') || 'light';
        applyTheme(savedTheme);
        themeToggle.addEventListener('click', () => applyTheme(htmlElement.getAttribute('data-bs-theme') === 'light' ? 'dark' : 'light'));
    </script>
