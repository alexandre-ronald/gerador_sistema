class RBACRuntimeError(PermissionError):
    pass


def _group_names(user):
    if user is None or not getattr(user, "is_authenticated", False):
        return set()
    if getattr(user, "is_superuser", False):
        return {"__superuser__"}
    return set(user.groups.values_list("name", flat=True))


def _role_ids_for_user(user):
    groups = _group_names(user)
    if "__superuser__" in groups:
        return set(RBAC_ROLES)
    return {role_id for role_id, role in RBAC_ROLES.items() if role["group"] in groups}


def has_entity_action(user, entity_name, action):
    if not RBAC_ENABLED:
        return True
    if user is not None and getattr(user, "is_superuser", False):
        return True
    policy = RBAC_ENTITIES.get(entity_name)
    if not policy:
        return False
    roles = _role_ids_for_user(user)
    return any(action in policy["roles"].get(role_id, []) for role_id in roles)


def require_entity_action(user, entity_name, action):
    if not has_entity_action(user, entity_name, action):
        raise RBACRuntimeError(f"Acesso negado para {entity_name}:{action}.")
    return True


def can_transition(user, entity_name, transition_id):
    if not RBAC_ENABLED:
        return True
    if user is not None and getattr(user, "is_superuser", False):
        return True
    policy = RBAC_ENTITIES.get(entity_name)
    if not policy:
        return False
    allowed_roles = set(policy["transitions"].get(transition_id, []))
    return bool(_role_ids_for_user(user) & allowed_roles)


def require_transition(user, entity_name, transition_id):
    if not can_transition(user, entity_name, transition_id):
        raise RBACRuntimeError(f"Transição não autorizada para {entity_name}:{transition_id}.")
    return True


def filter_transitions(user, entity_name, transitions):
    return [item for item in transitions if can_transition(user, entity_name, item.get("id"))]
