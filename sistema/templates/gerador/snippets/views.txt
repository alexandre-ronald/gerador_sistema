class {{ entidade.nome|title }}ListView(LoginRequiredMixin, PermissionRequiredMixin, ListView):

    def get_queryset(self):
        queryset = super().get_queryset()
        query = self.request.GET.get('q', '').strip()
        if query:
            search_filter = Q()
            {% for campo in entidade.campos.all %}
            {% if campo.tipo == 'CharField' or campo.tipo == 'TextField' or campo.tipo == 'EmailField' or campo.tipo == 'URLField' %}
            search_filter |= Q({{ campo.nome|lower }}__icontains=query)
            {% endif %}

            queryset = queryset.filter(search_filter)

        context['search_query'] = self.request.GET.get('q', '').strip()
