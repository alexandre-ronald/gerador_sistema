class {{ entidade.classe_nome }}ListView(LoginRequiredMixin, PermissionRequiredMixin, ListView):
    model = {{ entidade.classe_nome }}
    template_name = '{{ app_name }}/{{ entidade.codigo_nome }}_list.html'
    context_object_name = 'objetos'
    paginate_by = 10
    permission_required = '{{ app_name }}.view_{{ entidade.codigo_nome }}'
    raise_exception = True

    def get_queryset(self):
        queryset = super().get_queryset()
        query = self.request.GET.get('q', '').strip()
        if query:
            search_filter = Q()
{% for campo in entidade.campos_geracao %}{% if campo.tipo == 'CharField' or campo.tipo == 'TextField' or campo.tipo == 'EmailField' or campo.tipo == 'URLField' %}
            search_filter |= Q({{ campo.codigo_nome }}__icontains=query)
{% endif %}{% endfor %}
            queryset = queryset.filter(search_filter)
        return queryset

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['total_registros'] = self.get_queryset().count()
        context['titulo_pagina'] = "{{ entidade.nome }}"
        context['search_query'] = self.request.GET.get('q', '').strip()
        return context
