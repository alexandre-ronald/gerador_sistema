const charts={};
function escapeHtml(value){return String(value??'').replace(/[&<>'"]/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#039;','"':'&quot;'}[ch]));}
function widgetError(el,message){const old=el.querySelector('.dashboard-widget-error');if(old)old.remove();const node=document.createElement('div');node.className='dashboard-widget-error';node.innerHTML='<i class="bi bi-exclamation-circle-fill"></i><span>'+escapeHtml(message||'Não foi possível carregar este indicador.')+'</span>';el.appendChild(node);}
function renderWidget(item){const el=document.querySelector(`[data-widget-id="${CSS.escape(item.id)}"]`);if(!el)return;const oldError=el.querySelector('.dashboard-widget-error');if(oldError)oldError.remove();if(item.error){const value=el.querySelector('.widget-value-data');if(value)value.textContent='—';widgetError(el,item.error);return;}if(item.type==='metric'){const node=el.querySelector('.widget-value-data');if(node)node.textContent=item.value??'0';return;}if(item.type==='table'){const table=el.querySelector('table');if(!table)return;table.querySelector('thead').innerHTML=item.columns?.length?'<tr>'+item.columns.map(c=>`<th>${escapeHtml(c)}</th>`).join('')+'</tr>':'';table.querySelector('tbody').innerHTML=item.rows?.length?item.rows.map(r=>'<tr>'+r.map(v=>`<td>${escapeHtml(v??'')}</td>`).join('')+'</tr>').join(''):'<tr><td colspan="99"><div class="dashboard-table-empty"><i class="bi bi-inbox me-1"></i>Nenhum dado disponível.</div></td></tr>';return;}const canvas=el.querySelector('canvas');if(!canvas)return;if(charts[item.id])charts[item.id].destroy();charts[item.id]=new Chart(canvas,{type:item.type==='donut'?'doughnut':item.type==='pie'?'pie':item.type==='area'?'line':item.type,data:{labels:item.labels||[],datasets:[{label:item.title,data:item.values||[],fill:item.type==='area',tension:.34,borderWidth:2}]},options:{responsive:true,maintainAspectRatio:false,resizeDelay:50,plugins:{legend:{position:'bottom',labels:{usePointStyle:true,boxWidth:8,padding:16}}},scales:['pie','doughnut'].includes(item.type)?{}:{x:{grid:{display:false}},y:{beginAtZero:true,grid:{color:'rgba(148,163,184,.12)'}}}}});}
