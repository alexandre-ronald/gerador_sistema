def _allowed_items(request, items):
    user = getattr(request, "user", None)
    if user is None or not user.is_authenticated:
        return []
    return [item for item in items if not item["permission"] or user.has_perm(item["permission"])]


def _mark_active_items(request, items):
    resolver_match = getattr(request, "resolver_match", None)
    current_url_name = getattr(resolver_match, "url_name", None)
    current_app_name = getattr(resolver_match, "app_name", None)
    marked = []
    for item in items:
        active_names = item.get("active_url_names") or [item["url_name"].split(":")[-1]]
        marked.append({**item, "is_active": current_app_name == item.get("app_name") and current_url_name in active_names})
    return marked


def _current_navigation_context(request, modules):

    if current_app_name == "notifications":
        return {"module_label": "Comunicação", "item_label": "Notificações"}
    if current_url_name == "index":
        return {"module_label": "Início", "item_label": "Área de trabalho"}
    if current_url_name in {"dashboard", "dashboard_data"}:
        return {"module_label": "Análise", "item_label": "Dashboard"}
    if current_url_name in {"profile", "password_change"}:
        return {"module_label": "Conta", "item_label": "Meu perfil" if current_url_name == "profile" else "Alterar senha"}
    if current_url_name in {"user_list", "user_create", "user_update"}:
        return {"module_label": "Administração", "item_label": "Usuários"}
    if current_url_name == "role_list":
        return {"module_label": "Administração", "item_label": "Perfis de acesso"}
    for module in modules:
        for item in module["items"]:
            if item.get("is_active"):
                return {"module_label": "Relatórios" if item.get("is_report") else module["label"], "item_label": item["label"]}
    return {"module_label": "Área de trabalho", "item_label": "Visão geral"}


def _current_navigation_item(request):
    resolver_match = getattr(request, "resolver_match", None)
    current_url_name = getattr(resolver_match, "url_name", None)
    current_app_name = getattr(resolver_match, "app_name", None)
    if not current_url_name or not current_app_name:
        return None
    for module in NAVIGATION_MODULES:
        for item in module["items"]:
            if item.get("app_name") == current_app_name and current_url_name in (item.get("active_url_names") or []):
                return item
    return None


def _relation_filter_choices(request):
    resolver_match = getattr(request, "resolver_match", None)
    if not str(getattr(resolver_match, "url_name", "") or "").endswith("_list"):
        return []
    user = getattr(request, "user", None)
    if user is None or not user.is_authenticated:
        return []
    item = _current_navigation_item(request)
    if not item or item.get("is_notifications"):
        return []
    try:
        model = apps.get_model(item["app_name"], item["model_name"])
    except LookupError:
        return []
    if model is None:
        return []
    result = []
    for field in model._meta.fields:
        if not getattr(field, "is_relation", False) or (not getattr(field, "many_to_one", False) and not getattr(field, "one_to_one", False)):
            continue
        related_model = getattr(field.remote_field, "model", None)
        if related_model is None:
            continue
        options = [{"value": str(obj.pk), "label": str(obj)} for obj in related_model._default_manager.all()[:500]]
        options.sort(key=lambda option: option["label"].casefold())
        result.append({"field": field.name, "options": options})
    return result
